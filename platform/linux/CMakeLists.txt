cmake_minimum_required(VERSION 3.16)
project(hachimi_desktop C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(HACHIMI_USE_NEON "Use ARM NEON software blending" OFF)
option(HACHIMI_USE_PXP "Build the i.MX6ULL PXP adapter" OFF)
if(HACHIMI_USE_NEON)
    add_compile_definitions(HACHIMI_USE_NEON=1)
    add_compile_options(-mfpu=neon)
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "Linux LVGL configuration" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "Build LVGL demos" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "Build LVGL examples" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "Build ThorVG" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../lvgl" "${CMAKE_BINARY_DIR}/lvgl")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/hachimi_sources.cmake")
hachimi_add_runtime(hachimi_runtime "${CMAKE_CURRENT_SOURCE_DIR}/service_job.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/led_backend.c" lvgl::lvgl)
add_executable(hachimi_desktop main.c fb_transition.c)
if(HACHIMI_USE_PXP)
    target_sources(hachimi_desktop PRIVATE pxp/pxp_engine.c pxp/pxp_display.c)
    target_include_directories(hachimi_desktop PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/pxp/include")
    target_compile_definitions(hachimi_desktop PRIVATE HACHIMI_USE_PXP=1)
endif()
target_link_libraries(hachimi_desktop PRIVATE hachimi_runtime)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(hachimi_desktop PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
add_custom_target(copy_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${HACHIMI_ROOT}/custom/assets/ios" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/assets/ios"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${HACHIMI_ROOT}/custom/assets/maps" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/assets/maps")
add_custom_target(run COMMAND "$<TARGET_FILE:hachimi_desktop>"
    DEPENDS hachimi_desktop copy_assets WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
include(CTest)
if(BUILD_TESTING)
    hachimi_add_runtime(hachimi_preview_runtime "${CMAKE_CURRENT_SOURCE_DIR}/../preview/service_job.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../preview/led_backend.c" lvgl::lvgl)
    set_target_properties(hachimi_preview_runtime PROPERTIES EXCLUDE_FROM_ALL TRUE)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(HACHIMI_TEST_MUSIC_DIR "" CACHE PATH "Optional 12-track music smoke fixture")
    foreach(test architecture runtime clock calendar transition control_center maps messages_mail weather music)
        add_executable(${test}_smoke EXCLUDE_FROM_ALL "${HACHIMI_ROOT}/tests/${test}_smoke.c")
        target_link_libraries(${test}_smoke PRIVATE hachimi_preview_runtime)
        if(NOT test STREQUAL "music" OR HACHIMI_TEST_MUSIC_DIR)
            add_test(NAME ${test}_smoke COMMAND ${Python3_EXECUTABLE} "${HACHIMI_ROOT}/tests/run_smoke.py" "$<TARGET_FILE:${test}_smoke>")
            set_tests_properties(${test}_smoke PROPERTIES WORKING_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
            if(test STREQUAL "music")
                set_tests_properties(music_smoke PROPERTIES ENVIRONMENT "HACHIMI_MUSIC_DIR=${HACHIMI_TEST_MUSIC_DIR}")
            endif()
        endif()
        list(APPEND HACHIMI_SMOKE_TARGETS ${test}_smoke)
    endforeach()
    add_executable(led_service_smoke EXCLUDE_FROM_ALL "${HACHIMI_ROOT}/tests/led_service_smoke.c" "${HACHIMI_ROOT}/services/led.c")
    target_include_directories(led_service_smoke PRIVATE "${HACHIMI_ROOT}")
    add_test(NAME led_service_smoke COMMAND led_service_smoke)
    add_custom_target(smoke DEPENDS ${HACHIMI_SMOKE_TARGETS} led_service_smoke)
endif()
